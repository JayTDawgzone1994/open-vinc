# Framework output routing; only OSAL is required by this library.
add_library(vinc_output STATIC src/hub.cpp src/text_renderer.cpp)
add_library(vinc::output ALIAS vinc_output)
target_include_directories(vinc_output PUBLIC include)
target_link_libraries(vinc_output PUBLIC vinc::osal)
target_compile_features(vinc_output PUBLIC cxx_std_23)
if(MSVC)
    target_compile_options(vinc_output PRIVATE /W4 /utf-8)
else()
    target_compile_options(vinc_output PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
