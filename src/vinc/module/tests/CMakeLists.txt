add_executable(module_process_helper module_helper.cpp ../src/module_main.cpp)
target_link_libraries(module_process_helper PRIVATE module)
add_executable(module_integration_tests module_tests.cpp)
target_link_libraries(module_integration_tests PRIVATE module)
add_dependencies(module_integration_tests module_process_helper)
add_executable(silent_module_helper silent_helper.cpp)
target_link_libraries(silent_module_helper PRIVATE vinc::control)
add_dependencies(module_integration_tests silent_module_helper)
add_executable(control_channel_tests channel_tests.cpp)
target_link_libraries(control_channel_tests PRIVATE vinc::control)
foreach(target IN ITEMS module_process_helper silent_module_helper module_integration_tests control_channel_tests)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
add_test(NAME control.channel COMMAND control_channel_tests)
add_test(NAME module.process COMMAND module_integration_tests "$<TARGET_FILE:module_process_helper>" "$<TARGET_FILE:silent_module_helper>")
set_tests_properties(control.channel module.process PROPERTIES TIMEOUT 90)
