# Shared framework synchronization; native platform APIs remain in OSAL.
find_package(Threads REQUIRED)
add_library(vinc_sync STATIC src/interprocess_mutex.cpp)
add_library(vinc::sync ALIAS vinc_sync)
target_include_directories(vinc_sync PUBLIC include)
target_link_libraries(vinc_sync PUBLIC vinc::osal PRIVATE Threads::Threads)
target_compile_features(vinc_sync PUBLIC cxx_std_23)
if(MSVC)
    target_compile_options(vinc_sync PRIVATE /W4 /utf-8)
else()
    target_compile_options(vinc_sync PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
    add_executable(sync_interprocess_mutex_tests tests/interprocess_mutex_tests.cpp)
    target_link_libraries(sync_interprocess_mutex_tests PRIVATE vinc::sync Threads::Threads)
    if(MSVC)
        target_compile_options(sync_interprocess_mutex_tests PRIVATE /W4 /utf-8)
    else()
        target_compile_options(sync_interprocess_mutex_tests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME sync.interprocess_mutex COMMAND sync_interprocess_mutex_tests)
    set_tests_properties(sync.interprocess_mutex PROPERTIES TIMEOUT 90)
endif()
