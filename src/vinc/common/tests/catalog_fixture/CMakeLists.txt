cmake_minimum_required(VERSION 3.20)
project(error_catalog_fixture LANGUAGES CXX)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${COMMON_DIR}" common)
add_library(fixture INTERFACE)
target_include_directories(fixture INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(fixture INTERFACE vinc::common)
if(CASE STREQUAL "duplicate_code")
    target_compile_definitions(fixture INTERFACE DUPLICATE_CODE)
elseif(CASE STREQUAL "wrong_domain")
    target_compile_definitions(fixture INTERFACE WRONG_DOMAIN)
endif()
if(FIXTURE_ALTERNATE)
    target_compile_definitions(fixture INTERFACE EXPECT_UPDATED_CATALOG)
    vinc_register_error_catalog(TARGET fixture HEADER catalog.hpp SYMBOL fixture::updated_catalog)
else()
    vinc_register_error_catalog(TARGET fixture HEADER catalog.hpp SYMBOL fixture::catalog)
endif()
if(CASE STREQUAL "duplicate_registration")
    vinc_register_error_catalog(TARGET fixture HEADER catalog.hpp SYMBOL fixture::catalog)
elseif(CASE STREQUAL "duplicate_domain")
    vinc_register_error_catalog(TARGET fixture HEADER catalog.hpp SYMBOL fixture::duplicate_catalog)
endif()
if(NOT CASE STREQUAL "missing_finalize")
    vinc_finalize_error_catalog()
endif()
if(CASE STREQUAL "late_registration")
    vinc_register_error_catalog(TARGET fixture HEADER catalog.hpp SYMBOL fixture::duplicate_catalog)
elseif(CASE STREQUAL "duplicate_finalize")
    vinc_finalize_error_catalog()
endif()
add_executable(catalog_consumer main.cpp)
target_link_libraries(catalog_consumer PRIVATE vinc::errors fixture)
set_target_properties(catalog_consumer PROPERTIES
    CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
enable_testing()
add_test(NAME consumer COMMAND catalog_consumer)
