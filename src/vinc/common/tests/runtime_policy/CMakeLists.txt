cmake_minimum_required(VERSION 3.20)
project(runtime_policy_fixture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include("${POLICY}")
if(NOT CASE STREQUAL "standalone")
    set(VINC_DECLARE_PYTHON_RUNTIME TRUE)
endif()

function(build_library variant)
    vinc_runtime_name(library fixture ${variant})
    add_library(${library} STATIC library.cpp)
    vinc_apply_runtime(${library} ${variant})
endfunction()
vinc_runtime_variants(build_library)
if(CASE STREQUAL "standalone" AND TARGET fixture_python_runtime)
    message(FATAL_ERROR "Standalone component unexpectedly declares Python runtime libraries")
endif()

add_executable(consumer main.cpp)
if(CASE STREQUAL "native" OR CASE STREQUAL "standalone")
    vinc_apply_runtime(consumer NATIVE)
    target_compile_definitions(consumer PRIVATE EXPECTED_DYNAMIC=1)
    target_link_libraries(consumer PRIVATE fixture)
elseif(CASE STREQUAL "static")
    vinc_apply_runtime(consumer PYTHON)
    target_compile_definitions(consumer PRIVATE EXPECTED_DYNAMIC=0)
    vinc_runtime_link_libraries(consumer PYTHON PRIVATE fixture)
elseif(CASE STREQUAL "mixed")
    vinc_apply_runtime(consumer PYTHON)
    # An indirect native dependency must be rejected as well as a direct one.
    add_library(bridge INTERFACE)
    target_link_libraries(bridge INTERFACE fixture)
    target_link_libraries(consumer PRIVATE bridge fixture_python_runtime)
elseif(CASE STREQUAL "undeclared")
    add_library(foreign STATIC library.cpp)
    vinc_apply_runtime(consumer PYTHON)
    vinc_runtime_link_libraries(consumer PYTHON PRIVATE foreign)
elseif(CASE STREQUAL "invalid")
    vinc_apply_runtime(consumer TYPO)
else()
    message(FATAL_ERROR "Unknown fixture case")
endif()
set_target_properties(consumer PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
enable_testing()
add_test(NAME consumer COMMAND consumer)
