cmake_minimum_required(VERSION 3.20)
project(installed_python LANGUAGES CXX)
if(TEST_FOREIGN_PYTHON)
    # Simulate an unrelated consumer finding its own interpreter before VINC.
    set(Python_EXECUTABLE "${TEST_FOREIGN_PYTHON}")
    find_package(Python REQUIRED COMPONENTS Interpreter)
    set(foreign_python "${Python_EXECUTABLE}")
    get_target_property(foreign_location Python::Interpreter IMPORTED_LOCATION)
endif()
find_package(vinc CONFIG REQUIRED)
if(MSVC)
    foreach(target common module_runtime errors)
        get_target_property(crt vinc::${target}_python_runtime INTERFACE_VINC_CRT)
        if(NOT crt STREQUAL "static")
            message(FATAL_ERROR "Installed Python target lost its CRT contract: ${target}")
        endif()
    endforeach()
    if(TEST_MIXED_CRT)
        add_library(mixed STATIC peer.cpp)
        target_link_libraries(mixed PRIVATE vinc::common vinc::common_python_runtime)
        return()
    endif()
endif()
vinc_add_app(PythonApp INSTANCE_NAME sdk-python VINC_BUILD_CONFIG vinc.conf)
if(TEST_FOREIGN_PYTHON)
    if(NOT Python_EXECUTABLE STREQUAL foreign_python)
        message(FATAL_ERROR "VINC changed the unrelated consumer's Python_EXECUTABLE")
    endif()
    get_target_property(location Python::Interpreter IMPORTED_LOCATION)
    if(NOT location STREQUAL foreign_location)
        message(FATAL_ERROR "VINC changed the unrelated consumer's interpreter target")
    endif()
    get_target_property(binding PythonApp VINC_APP_PYTHON_TARGET)
    if(binding)
        get_target_property(interpreter ${binding} VINC_PYTHON_ABI_EXECUTABLE)
        get_filename_component(expected "${Python3_EXECUTABLE}" ABSOLUTE)
        if(NOT interpreter STREQUAL expected)
            message(FATAL_ERROR "Binding did not use the fixture's Kconfig interpreter")
        endif()
    endif()
endif()
# Late registration must reach the Python extension too.
add_message_dir(messages)
add_module(python_peer MODULE_TYPE_NAME NativePeer SOURCES peer.cpp)
set_target_properties(PythonApp PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/application output")
