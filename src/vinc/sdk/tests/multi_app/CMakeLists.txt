cmake_minimum_required(VERSION 3.20)
project(multi_app LANGUAGES CXX)
if(VINC_SOURCE)
    set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
    set(VINC_BUILD_APPS OFF CACHE BOOL "" FORCE)
    set(VINC_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(VINC_BUILD_PYTHON OFF CACHE BOOL "" FORCE)
    set(VINC_BUILD_UI ON CACHE BOOL "" FORCE)
    add_subdirectory("${VINC_SOURCE}/src" framework)
else()
    find_package(vinc CONFIG REQUIRED)
endif()

function(fixture_app name identity type)
    set(APP_NAME "${name}")
    set(MODULE_TYPE "${type}")
    configure_file("${PROJECT_SOURCE_DIR}/startup.tcl.in" "${CMAKE_CURRENT_BINARY_DIR}/startup.tcl" @ONLY)
    vinc_add_app(${name} INSTANCE_NAME "${identity}"
        VINC_BUILD_CONFIG app.config STARTUP_SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/startup.tcl")
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/${name}-$<CONFIG>.path" CONTENT "$<TARGET_FILE:${name}>")
endfunction()

# App declarations precede sibling Module and schema registration deliberately.
add_subdirectory(first)
add_subdirectory(second)
add_subdirectory(third)
add_message_dir(messages)
foreach(type FirstModule SecondModule SharedModule)
    add_module(${type} MODULE_TYPE_NAME ${type} SOURCES module.cpp)
    target_compile_definitions(${type} PRIVATE MODULE_KIND="${type}")
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/${type}-$<CONFIG>.path" CONTENT "$<TARGET_FILE:${type}>")
endforeach()
add_module(Unselected MODULE_TYPE_NAME Unselected SOURCES forbidden.cpp)
# Explicit fault injection is not a registered or selectable ModuleType.
vinc_add_module_plugin(StalledOfficial module.cpp)
set_property(TARGET StalledOfficial PROPERTY EXCLUDE_FROM_ALL TRUE)
target_link_libraries(StalledOfficial PRIVATE vinc::sdk vinc::messages)
target_compile_definitions(StalledOfficial PRIVATE MODULE_KIND="StalledOfficial" STALL_BEFORE_INJECTION)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/StalledOfficial-$<CONFIG>.path"
    CONTENT "$<TARGET_FILE:StalledOfficial>")
if(RESERVED_TYPE)
    add_module(Invalid MODULE_TYPE_NAME webUi SOURCES module.cpp)
endif()

function(record_composition)
    foreach(app First Second Third)
        get_target_property(selected ${app} VINC_APP_MODULE_TARGETS)
        get_target_property(generated ${app} VINC_APP_GENERATED)
        get_target_property(python_target ${app} VINC_APP_PYTHON_TARGET)
        file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/${app}-$<CONFIG>.selection" CONTENT "${selected}")
        file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/${app}-$<CONFIG>.generated" CONTENT "${generated}")
        file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/${app}-$<CONFIG>.python" CONTENT "${python_target}")
    endforeach()
    if(TARGET vinc_python_native)
        get_target_property(package vinc_python_native VINC_PYTHON_DIRECTORY)
        get_target_property(resources vinc_python_native VINC_PYTHON_RESOURCES)
        string(REPLACE ";" "\n" resources "${resources}")
        file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/python-$<CONFIG>.path" CONTENT "${package}")
        file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/python-$<CONFIG>.resources" CONTENT "${resources}")
        file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/native-$<CONFIG>.path" CONTENT "$<TARGET_FILE:vinc_python_native>")
    endif()
endfunction()
cmake_language(DEFER CALL record_composition)
