cmake_minimum_required(VERSION 3.20)
project(installed_application LANGUAGES CXX)
find_package(vinc CONFIG REQUIRED)
add_message_dir(messages)
add_message_dir(messages/.)
add_module(app_sender MODULE_TYPE_NAME AppSender SOURCES module.cpp)
target_compile_definitions(app_sender PRIVATE TEST_SENDER)
add_module(app_reader MODULE_TYPE_NAME AppReader SOURCES module.cpp)
if(TEST_DUPLICATE_MODULE)
    add_module(duplicate MODULE_TYPE_NAME AppReader SOURCES module.cpp)
endif()
set(options)
if(EMBED_STARTUP)
    list(APPEND options STARTUP_SCRIPT startup.tcl)
endif()
vinc_add_app(PackageApp INSTANCE_NAME sdk-package-app VINC_BUILD_CONFIG app.config ${options})
if(TEST_SECOND_APP)
    vinc_add_app(Second INSTANCE_NAME sdk-second VINC_BUILD_CONFIG app.config)
endif()
# Registration after app creation must still be included at configure completion.
add_message_dir(late_messages)
if(TEST_MISSING_MESSAGES)
    add_message_dir(missing)
endif()
# Exercise output changes after helper invocation and keep cwd unrelated at run time.
set_target_properties(PackageApp PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/application output")
