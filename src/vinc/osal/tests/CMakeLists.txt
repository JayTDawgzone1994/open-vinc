    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/test programs")
add_executable(osal_process_tests process_tests.cpp)
target_link_libraries(osal_process_tests PRIVATE vinc::osal Threads::Threads)

    target_compile_options(osal_process_helper PRIVATE /W4 /utf-8)
    target_compile_options(osal_process_tests PRIVATE /W4 /utf-8)
else()
    target_compile_options(osal_process_helper PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(osal_process_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME osal.process COMMAND osal_process_tests "$<TARGET_FILE:osal_process_helper>")
set_tests_properties(osal.process PROPERTIES TIMEOUT 60)
